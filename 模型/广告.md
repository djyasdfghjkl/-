为了帮助你实现用户在线时长和广告行为统计的后端需求，我为你设计了一套完整的方案，使用 **Node.js + Koa** 框架，数据库采用 **MySQL**。这套方案包含了详细的接口定义、数据表结构、业务逻辑代码以及防作弊设计。

---

## 一、技术栈准备

- **Node.js** (v14+)
- **Koa** (v2)
- **mysql2/promise** (支持 async/await 的 MySQL 驱动)
- **uuid** (生成会话 ID)
- **koa-body** (解析请求体)
- **koa-router** (路由管理)

安装依赖：

```bash
npm install koa koa-router koa-body mysql2 uuid
```

## 二、数据表设计 (MySQL)

### 1. 用户会话表 `user_session`

```sql
CREATE TABLE `user_session` (
  `id` bigint(20) NOT NULL AUTO_INCREMENT,
  `openid` varchar(64) NOT NULL COMMENT '用户openid',
  `session_id` varchar(64) NOT NULL COMMENT '会话唯一ID',
  `enter_time` datetime NOT NULL COMMENT '进入时间',
  `leave_time` datetime DEFAULT NULL COMMENT '离开时间',
  `duration` int(11) DEFAULT '0' COMMENT '会话时长(秒)',
  `enter_page` varchar(128) DEFAULT NULL COMMENT '进入页面路径',
  `ip` varchar(45) DEFAULT NULL COMMENT '客户端IP',
  `user_agent` varchar(256) DEFAULT NULL COMMENT '设备信息',
  `create_time` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_session_id` (`session_id`),
  KEY `idx_openid` (`openid`),
  KEY `idx_create_time` (`create_time`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户会话记录';
```

### 2. 广告行为表 `ad_action`

```sql
CREATE TABLE `ad_action` (
  `id` bigint(20) NOT NULL AUTO_INCREMENT,
  `openid` varchar(64) NOT NULL,
  `session_id` varchar(64) NOT NULL,
  `ad_unit_id` varchar(64) NOT NULL COMMENT '广告单元ID',
  `ad_type` varchar(20) NOT NULL COMMENT '广告类型：banner,激励视频,插屏,原生模板',
  `action_type` varchar(20) NOT NULL COMMENT '行为类型：request,exposure,click,play_complete',
  `action_time` datetime NOT NULL COMMENT '行为发生时间',
  `ext_info` json DEFAULT NULL COMMENT '扩展信息',
  `ip` varchar(45) DEFAULT NULL,
  `create_time` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_openid` (`openid`),
  KEY `idx_session` (`session_id`),
  KEY `idx_action_time` (`action_time`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='广告行为记录';
```

### 3. 用户行为日汇总表 `user_stats_daily`

```sql
CREATE TABLE `user_stats_daily` (
  `id` bigint(20) NOT NULL AUTO_INCREMENT,
  `openid` varchar(64) NOT NULL,
  `stat_date` date NOT NULL COMMENT '统计日期',
  `total_duration` int(11) DEFAULT '0' COMMENT '总在线时长(秒)',
  `session_count` int(11) DEFAULT '0' COMMENT '会话次数',
  `ad_request_count` int(11) DEFAULT '0' COMMENT '广告请求次数',
  `ad_exposure_count` int(11) DEFAULT '0' COMMENT '广告曝光次数',
  `ad_click_count` int(11) DEFAULT '0' COMMENT '广告点击次数',
  `ad_play_complete_count` int(11) DEFAULT '0' COMMENT '激励视频完整播放次数',
  `update_time` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_openid_date` (`openid`,`stat_date`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户每日汇总';
```

### 2. 控制器 `controllers/stats.js`

```javascript
const pool = require("../models/db");
const { v4: uuidv4 } = require("uuid");

// 会话开始
async function sessionStart(ctx) {
  const { openid, enter_page = "" } = ctx.request.body;
  const session_id = uuidv4();
  const ip = ctx.request.ip;
  const user_agent = ctx.headers["user-agent"];

  if (!openid) {
    ctx.status = 400;
    ctx.body = { code: 1, msg: "openid required" };
    return;
  }

  const connection = await pool.getConnection();
  try {
    await connection.execute(
      `INSERT INTO user_session (openid, session_id, enter_time, enter_page, ip, user_agent)
       VALUES (?, ?, NOW(), ?, ?, ?)`,
      [openid, session_id, enter_page, ip, user_agent],
    );
    ctx.body = { code: 0, msg: "success", data: { session_id } };
  } catch (err) {
    console.error(err);
    ctx.status = 500;
    ctx.body = { code: 2, msg: "server error" };
  } finally {
    connection.release();
  }
}

// 会话结束
async function sessionEnd(ctx) {
  const { session_id, openid } = ctx.request.body;
  if (!session_id) {
    ctx.status = 400;
    ctx.body = { code: 1, msg: "session_id required" };
    return;
  }

  const connection = await pool.getConnection();
  try {
    // 更新 leave_time 和 duration
    const [result] = await connection.execute(
      `UPDATE user_session 
       SET leave_time = NOW(), duration = TIMESTAMPDIFF(SECOND, enter_time, NOW())
       WHERE session_id = ? AND leave_time IS NULL`,
      [session_id],
    );

    if (result.affectedRows === 0) {
      ctx.body = { code: 3, msg: "session not found or already ended" };
      return;
    }

    // 获取刚更新的记录用于汇总
    const [rows] = await connection.execute(
      `SELECT openid, duration FROM user_session WHERE session_id = ?`,
      [session_id],
    );
    if (rows.length > 0) {
      const { openid: userOpenid, duration } = rows[0];
      // 异步更新日汇总表（可以先不等待，或者使用Promise）
      updateUserDailyStats(userOpenid, { duration, session_count: 1 });
    }

    ctx.body = { code: 0, msg: "success" };
  } catch (err) {
    console.error(err);
    ctx.status = 500;
    ctx.body = { code: 2, msg: "server error" };
  } finally {
    connection.release();
  }
}

// 广告行为上报
async function adAction(ctx) {
  const {
    openid,
    session_id,
    ad_unit_id,
    ad_type,
    action_type,
    action_time,
    ext_info,
  } = ctx.request.body;
  const ip = ctx.request.ip;

  if (!openid || !session_id || !ad_unit_id || !ad_type || !action_type) {
    ctx.status = 400;
    ctx.body = { code: 1, msg: "missing required fields" };
    return;
  }

  // 可选：校验行为顺序（request -> exposure -> click/play_complete）
  // 简单示例：检查session是否存在
  const connection = await pool.getConnection();
  try {
    // 验证会话是否存在（防止伪造）
    const [sessions] = await connection.execute(
      "SELECT openid FROM user_session WHERE session_id = ?",
      [session_id],
    );
    if (sessions.length === 0) {
      ctx.body = { code: 3, msg: "invalid session_id" };
      return;
    }

    const actionTime =
      action_time || new Date().toISOString().slice(0, 19).replace("T", " ");
    await connection.execute(
      `INSERT INTO ad_action (openid, session_id, ad_unit_id, ad_type, action_type, action_time, ext_info, ip)
       VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
      [
        openid,
        session_id,
        ad_unit_id,
        ad_type,
        action_type,
        actionTime,
        JSON.stringify(ext_info || {}),
        ip,
      ],
    );

    // 异步更新日汇总表
    updateUserDailyStats(openid, { [action_type]: 1 });

    ctx.body = { code: 0, msg: "success" };
  } catch (err) {
    console.error(err);
    ctx.status = 500;
    ctx.body = { code: 2, msg: "server error" };
  } finally {
    connection.release();
  }
}

// 用户统计查询
async function getUserStats(ctx) {
  const { openid, start_date, end_date } = ctx.query;
  if (!openid) {
    ctx.status = 400;
    ctx.body = { code: 1, msg: "openid required" };
    return;
  }

  let sql = "SELECT * FROM user_stats_daily WHERE openid = ?";
  const params = [openid];
  if (start_date && end_date) {
    sql += " AND stat_date BETWEEN ? AND ?";
    params.push(start_date, end_date);
  } else {
    // 默认最近7天
    sql += " AND stat_date >= DATE_SUB(CURDATE(), INTERVAL 7 DAY)";
  }
  sql += " ORDER BY stat_date DESC";

  const connection = await pool.getConnection();
  try {
    const [rows] = await connection.execute(sql, params);
    // 聚合计算
    const total = rows.reduce(
      (acc, row) => {
        acc.total_duration += row.total_duration;
        acc.session_count += row.session_count;
        acc.ad_request_count += row.ad_request_count;
        acc.ad_exposure_count += row.ad_exposure_count;
        acc.ad_click_count += row.ad_click_count;
        acc.ad_play_complete_count += row.ad_play_complete_count;
        return acc;
      },
      {
        total_duration: 0,
        session_count: 0,
        ad_request_count: 0,
        ad_exposure_count: 0,
        ad_click_count: 0,
        ad_play_complete_count: 0,
      },
    );

    const avg_duration_per_day = rows.length
      ? total.total_duration / rows.length
      : 0;

    ctx.body = {
      code: 0,
      data: {
        ...total,
        avg_duration_per_day: Math.round(avg_duration_per_day),
        daily_stats: rows,
      },
    };
  } catch (err) {
    console.error(err);
    ctx.status = 500;
    ctx.body = { code: 2, msg: "server error" };
  } finally {
    connection.release();
  }
}

// 心跳接口（可选）
async function heartbeat(ctx) {
  const { session_id } = ctx.request.body;
  if (!session_id) {
    ctx.status = 400;
    ctx.body = { code: 1, msg: "session_id required" };
    return;
  }

  const connection = await pool.getConnection();
  try {
    // 更新会话的最后心跳时间，可用于检测超时会话
    await connection.execute(
      `UPDATE user_session SET last_heartbeat = NOW() WHERE session_id = ?`,
      [session_id],
    );
    ctx.body = { code: 0, msg: "ok" };
  } catch (err) {
    console.error(err);
    ctx.status = 500;
    ctx.body = { code: 2, msg: "server error" };
  } finally {
    connection.release();
  }
}

// ---------- 辅助函数：异步更新日汇总表 ----------
async function updateUserDailyStats(openid, stats) {
  const date = new Date().toISOString().slice(0, 10); // YYYY-MM-DD
  const connection = await pool.getConnection();
  try {
    // 构建动态更新字段
    const fields = [];
    const values = [];
    if (stats.duration) {
      fields.push("total_duration = total_duration + ?");
      values.push(stats.duration);
    }
    if (stats.session_count) {
      fields.push("session_count = session_count + ?");
      values.push(stats.session_count);
    }
    if (stats.request) {
      fields.push("ad_request_count = ad_request_count + ?");
      values.push(stats.request);
    }
    if (stats.exposure) {
      fields.push("ad_exposure_count = ad_exposure_count + ?");
      values.push(stats.exposure);
    }
    if (stats.click) {
      fields.push("ad_click_count = ad_click_count + ?");
      values.push(stats.click);
    }
    if (stats.play_complete) {
      fields.push("ad_play_complete_count = ad_play_complete_count + ?");
      values.push(stats.play_complete);
    }

    if (fields.length === 0) return;

    const sql = `
      INSERT INTO user_stats_daily (openid, stat_date, ${fields.map((f) => f.split(" ")[0]).join(", ")})
      VALUES (?, ?, ${values.map(() => "?").join(", ")})
      ON DUPLICATE KEY UPDATE ${fields.join(", ")}
    `;

    await connection.execute(sql, [openid, date, ...values]);
  } catch (err) {
    console.error("updateUserDailyStats error:", err);
  } finally {
    connection.release();
  }
}

module.exports = {
  sessionStart,
  sessionEnd,
  adAction,
  getUserStats,
  heartbeat,
};
```

## 五、防作弊与数据一致性设计

### 1. 防重复上报

- 可以为每个广告行为在前端生成唯一请求ID（`request_id`），后端在插入前检查是否已存在，实现幂等性。
- 或者通过`action_time`+`ad_unit_id`+`action_type`+`session_id`组合去重。

### 2. 行为顺序校验

- 在插入`ad_action`前，检查该会话是否已有`request`记录，然后才能有`exposure`等，防止恶意直接上报点击。

### 3. 会话超时自动关闭

- 定期执行定时任务（如每5分钟），扫描`last_heartbeat`超过一定时间（如10分钟）且`leave_time`为空的会话，自动调用`sessionEnd`逻辑。

### 4. 数据汇总异步化

- 使用消息队列（如RabbitMQ、Kafka）处理`ad_action`和`session_end`事件，避免同步更新汇总表造成的性能瓶颈。
- 当前示例采用了异步函数调用，实际生产环境建议使用消息队列。

### 5. 风控规则

- 同一IP/设备短时间大量请求需限流。
- 检查广告行为频率，例如单个用户每分钟点击超过一定阈值（如20次）可标记异常。
- 每日汇总数据可设置上限，如`total_duration`不超过24小时，防止刷时长。
